\noindent
$u: [0, T] \times \partial C \to \R$ represents receptors, $u_b: [0, T] \times \partial C \to \R$ complex, $v: [0, T] \times E \to \R$ ligand. $k_1, k_2 \in \R$ reaction coefficients, $d_1, d_2 \in \R$ decay coefficients, $D \in \R$ diffusion coefficient, $p_1, p_2$ some production functions. ($p_1(u) = au,\ p_2(v) = bv$?)

\begin{equation*}
    \begin{aligned}
        \partial_t u &= - k_1 u^2 v + k_2 u_b + p_1(u) - d_1 u & &\text{on } (0, T) \times \partial C \\
        \partial_t u_b &= k_1 u^2 v - k_2 u_b - d_2 u_b & &\text{on } (0, T) \times \partial C \\
        \partial_t v &= D \Delta v + p_2(v) & &\text{on } (0, T) \times E \\
        D \partial_\nu v &= - k_1 u^2 v + k_2 u_b & &\text{on } [0, T) \times \partial C \\
        D \partial_\nu v &= 0 & &\text{on } [0, T) \times \partial \Omega \\
    \end{aligned}
\end{equation*}
