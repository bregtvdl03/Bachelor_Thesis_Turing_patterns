Next, we develop a weakened and more general notion of solutions to ODEs.

\begin{definition}{}{}
    Let $n$ be a positive integer, $I$ an interval, $x: I \to \R^n$ and $f: S \to \R^n$ for some set $S \subseteq \R \times \R^n$. Then $x$ is called a \emph{solution of}
%
    \begin{equation}\label{eq:standard_ode}
        x'(t) = f(t, x(t))
    \end{equation}

    \emph{in the sense of Carath\'eodory on $I$} if $x$ is absolutely continuous, $(t, x(t)) \in S$ for every $t \in I$ and \cref{eq:standard_ode} is satisfied for almost every $t \in I$.
\end{definition}

\noindent
Since this definition requires less from the solution, there is also a more general existence theorem, found in \cite{coddington_theory_1984}.

\begin{theorem}{Carath\'eodory's existence theorem}{caratheodory}
    Let $n$ be a positive integer, $a, b > 0$ and $(t_0, x_0) \in \R \times \R^n$. Define
%
    \begin{equation*}
        R := \{ (t, x) \in \R \times \R^n \mid |t - t_0| \leq a \text{ and } \left|x - x_0\right| \leq b \}
    \end{equation*}

    and let $f: R \to \R^n$ be continuous in $x$ for every fixed $t$, measurable in $t$ for every fixed $x$ and bounded by some time-dependent integrable function $m$,
    %
    \begin{equation*}
        \left|f(t, x)\right| \leq m(t)
    \end{equation*}

    for all $(t, x) \in R$. Then the differential equation
%
    \begin{equation*}
        x'(t) = f(t, x(t))
    \end{equation*}

    has a solution in the sense of Carath\'eodory on the interval $|t - t_0| \leq \beta$ for some $\beta > 0$, satisfying $x(0) = x_0$.
\end{theorem}
